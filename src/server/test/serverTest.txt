# Health check
curl http://localhost:7654/health

# Init session
curl -s -X POST http://localhost:7654/session/init \
  -H "Content-Type: application/json" \
  -d '{"clientId":"test-cli","workspaceRoot":"/tmp/testproject"}' | python3 -m json.tool

# Open a document
curl -s -X POST http://localhost:7654/document/open \
  -H "Content-Type: application/json" \
  -d '{
    "sessionId":"SAMPLE-ID",
    "filePath":"/tmp/testproject/app.js",
    "content":"function main() {\n  console.log(\"hello\");\n}\n"
  }' | python3 -m json.tool

# Toggle debug mode ON
curl -s -X POST http://localhost:7654/mode/toggle \
  -H "Content-Type: application/json" \
  -d '{"sessionId":"SAMPLE-ID"}' | python3 -m json.tool

# Toggle insert mode to DEBUG
curl -s -X POST http://localhost:7654/mode/insert-toggle \
  -H "Content-Type: application/json" \
  -d '{"sessionId":"SAMPLE-ID"}' | python3 -m json.tool

# Make an edit (insert debug code at offset 20)
curl -s -X POST http://localhost:7654/document/change \
  -H "Content-Type: application/json" \
  -d '{
    "sessionId":"SAMPLE-ID",
    "filePath":"/tmp/testproject/app.js",
    "changes":[{"rangeOffset":20,"rangeLength":0,"text":"  // DEBUG LINE\n"}]
  }' | python3 -m json.tool

# Toggle debug mode OFF — the debug line should disappear from displayContent
curl -s -X POST http://localhost:7654/mode/toggle \
  -H "Content-Type: application/json" \
  -d '{"sessionId":"SAMPLE-ID"}' | python3 -m json.tool
